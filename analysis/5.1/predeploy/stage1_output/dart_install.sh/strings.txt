# Check if the package architecture matches the device architecture
checkArchitectureCompatibility() 
    VPNARCH=x86_64 
    # Check the current device architecture
    ARCH=$(uname -m)
    if [ "$ARCH" != "x86_64" ] && [ "$ARCH" != "aarch64" ]; then
        echo "Unknown architecture: $ARCH"
        exit 1
    # Compare the VPN architecture with the device architecture
    if [ "$VPNARCH" != "$ARCH" ]; then
        echo "Architecture mismatch: Installer is for $VPNARCH, but the device is $ARCH."
        echo "Installation aborted due to architecture mismatch!" >> /tmp/${LOGFNAME}
        exit 1
csc_vercmp() 
    ver_from="${1:?missing from-version}"
    ver_to="${2:?missing to-version}"
    if [ "$ver_from" = "$ver_to" ]; then
        echo same
    else
        ver_min="$(printf '%s\n' "$ver_from" "$ver_to" | sort -V | head -n1)"
        if [ "$ver_min" = "$ver_from" ]; then
            echo older
        else
            echo newer
        fi
checkInstalledVersion()
  failed=false
  # version of DART being installed has to be greater than the current DART version
  if [ -f ${DARTMANIFEST} ]; then
      DARTPREVVERSION=$(awk -F"\"" '/file version/ { print $2 }' ${DARTMANIFEST})
      DARTCURRVERSION=5.1.12.146
      existing_ver="$(csc_vercmp "$DARTPREVVERSION" "$DARTCURRVERSION")"
      if [ "$existing_ver" != older ]; then
        failed=true
      fi
  if [ "$failed" = true ]; then
    if [ "$existing_ver" = same ]; then
      echo "Version ${DARTCURRVERSION} is already installed!"
      echo "Version ${DARTCURRVERSION} is already installed!" >> /tmp/${LOGFNAME}
    else
      echo "A higher version ${DARTPREVVERSION} of DART is already installed!"
      echo "A higher version ${DARTPREVVERSION} of DART is already installed!" >> /tmp/${LOGFNAME}
    echo "Exiting now."
    echo "Exiting now." >> /tmp/${LOGFNAME}
    exit 1
